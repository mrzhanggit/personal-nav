👹 BOSS 宣言 · 需求书

  • 用面向对象做一个双人卡牌对战游戏，需要四个类：
  • Card 类：一张卡牌（名字、攻击力），能被漂亮地打印和比较
  • Deck 类：牌库（生成 24 张牌、洗牌、抽牌）
  • Player 类：玩家（名字、HP、手牌、摸牌、出牌、挨打）
  • Game 类：裁判（回合制循环、伤害结算、胜负判定）
  • 综合运用：类 / 继承 / 魔法方法 / 列表 / 随机模块
  • 通关奖励：BOSS 掉落 「系统设计师」称号 🏆

    2🔨 分步搭建 · 六步斩杀 BOSS

    每一小步都是可以独立运行验证的代码。跟着敲，每敲完一步就在电脑上跑一遍，确认输出一致再进下一步。

    第 1 步 · Card 类：让一张牌能"自我介绍"

    先用 __init__ 存数据，再配两个魔法方法：__str__ 决定 print 时的样子，__eq__ 决定 == 怎么比较。

    class Card: def __init__(self, name, attack): self.name = name # 卡牌名字 self.attack = attack # 攻击力 def __str__(self): # print(卡牌) 时调用 return "%s(攻%d)" % (self.name, self.attack) __repr__ = __str__ # 放进列表打印时也好看 def __eq__(self, other): # 定义 == 的比较规则 if not isinstance(other, Card): return NotImplemented return self.name == other.name and self.attack == other.attack card = Card("炎龙", 6) print(card) print(card == Card("炎龙", 6), card == Card("火精灵", 1))
    炎龙(攻6) True False

    💡 没有 __eq__ 时，card == Card("炎龙", 6) 比较的是两个对象的内存地址，结果会是 False——这是第 18 课讲过的坑。

    第 2 步 · Deck 类：发牌员上岗（洗牌 + 抽牌）

    牌库本质是一个列表。洗牌用 random.shuffle；抽牌用 list.pop() 从尾部弹出，一步到位还高效。

    import random class Deck: POOL = [("火精灵", 1), ("石盾卫", 2), ("雷刃客", 3), ("暗影刺客", 4), ("圣光骑士", 5), ("炎龙", 6)] def __init__(self): self.cards = [] for name, attack in self.POOL: for _ in range(4): # 每种牌 4 张，共 24 张 self.cards.append(Card(name, attack)) self.shuffle() # 建好就洗牌 def shuffle(self): random.shuffle(self.cards) # 原地打乱列表 def draw(self): if not self.cards: # 牌库空了 return None return self.cards.pop() # 从牌库顶抽走一张 def __len__(self): # len(牌库) 直接看剩几张 return len(self.cards) random.seed(1) # 固定种子，方便核对输出 deck = Deck() print("牌库张数：", len(deck)) print("抽到：", deck.draw(), "/", deck.draw()) print("剩：", len(deck), "张")
    牌库张数： 24 抽到： 石盾卫(攻2) / 圣光骑士(攻5) 剩： 22 张
    第 3 步 · Player 类：战士的诞生

    玩家有名字、HP 和一手牌。出牌策略先用最简单的：永远打出攻击力最高的牌（用 max + remove 组合拳）。

    class Player: def __init__(self, name, hp=30): self.name = name self.hp = hp self.hand = [] # 手牌列表 def draw_from(self, deck, n=1): # 摸 n 张牌 for _ in range(n): card = deck.draw() if card is not None: # 牌库空了就摸不到 self.hand.append(card) def play_best(self): # 打出攻击力最高的牌 if not self.hand: return None best = max(self.hand, key=lambda c: c.attack) self.hand.remove(best) return best def take_damage(self, dmg): # 挨打，血量不低于 0 self.hp = max(0, self.hp - dmg) def is_alive(self): return self.hp > 0 hero = Player("阿呆") hero.draw_from(deck, 3) # 接着第 2 步的 deck 摸 3 张 print(hero.name, "HP:", hero.hp) print("手牌：", hero.hand) print("打出：", hero.play_best()) hero.take_damage(6) print("挨了 6 点伤害，HP:", hero.hp)
    阿呆 HP: 30 手牌： [火精灵(攻1), 雷刃客(攻3), 火精灵(攻1)] 打出： 雷刃客(攻3) 挨了 6 点伤害，HP: 24

    💡 注意 print(手牌列表) 时，列表里的每张牌走的是 __repr__——这就是第 1 步里 __repr__ = __str__ 的功劳。

    第 4 步 · Game 类：裁判进场

    回合循环写在 Game 而不是 Player 里：玩家只管自己的动作，规则和节奏归裁判管。每个回合双方各行动一次：抽 1 张 → 打出最强牌 → 对方扣血。

    class Game: def __init__(self, p1, p2): self.players = [p1, p2] # 两名玩家 self.deck = Deck() # 双方共用一个牌库 self.round_no = 0 def start(self): # 开局：起手各抽 3 张 for p in self.players: p.draw_from(self.deck, 3) def one_turn(self, attacker, defender): """一个玩家的一回合：抽 1 张 → 打出最强牌""" attacker.draw_from(self.deck, 1) card = attacker.play_best() if card is None: print("%s 没牌可出，只能干瞪眼！" % attacker.name) return defender.take_damage(card.attack) print("%s 打出 %s，对 %s 造成 %d 点伤害（%s 剩余 HP %d）" % (attacker.name, card, defender.name, card.attack, defender.name, defender.hp)) def battle(self): # 回合循环 + 胜负判定 print("=== 卡牌对战：%s VS %s ===" % (self.players[0].name, self.players[1].name)) self.start() p1, p2 = self.players winner = None while True: self.round_no += 1 print("—— 第 %d 回合（牌库剩 %d 张）——" % (self.round_no, len(self.deck))) for attacker, defender in ((p1, p2), (p2, p1)): # 各行动一次 self.one_turn(attacker, defender) if not defender.is_alive(): # 有人倒下！ winner = attacker break if winner: break if len(self.deck) == 0 and all(len(p.hand) == 0 for p in self.players): print("双方都没牌了，握手言和，平局！") # 平局出口 return None print("🏆 %s 获胜！大战 %d 回合" % (winner.name, self.round_no)) return winner

    第 6 步 · 完整合并代码（约 120 行 · 可直接运行）

    把前 5 步合并成一个完整文件 card_battle.py，从上往下就是最终答案。运行输出与第 5 步完全一致（seed=42 固定战局）。

    # card_battle.py —— BOSS 战完整答案（Python 3 可直接运行） import random class Card: """一张卡牌：名字 + 攻击力""" def __init__(self, name, attack): self.name = name self.attack = attack def __str__(self): return "%s(攻%d)" % (self.name, self.attack) __repr__ = __str__ def __eq__(self, other): if not isinstance(other, Card): return NotImplemented return self.name == other.name and self.attack == other.attack class Deck: """牌库：生成 24 张牌，支持洗牌和抽牌""" POOL = [("火精灵", 1), ("石盾卫", 2), ("雷刃客", 3), ("暗影刺客", 4), ("圣光骑士", 5), ("炎龙", 6)] def __init__(self): self.cards = [] for name, attack in self.POOL: for _ in range(4): self.cards.append(Card(name, attack)) self.shuffle() def shuffle(self): random.shuffle(self.cards) def draw(self): if not self.cards: return None return self.cards.pop() def __len__(self): return len(self.cards) class Player: """玩家：HP、手牌、摸牌出牌""" def __init__(self, name, hp=30): self.name = name self.hp = hp self.hand = [] def draw_from(self, deck, n=1): for _ in range(n): card = deck.draw() if card is not None: self.hand.append(card) def play_best(self): if not self.hand: return None best = max(self.hand, key=lambda c: c.attack) self.hand.remove(best) return best def take_damage(self, dmg): self.hp = max(0, self.hp - dmg) def is_alive(self): return self.hp > 0 class Game: """裁判：组织回合、结算伤害、判定胜负""" def __init__(self, p1, p2): self.players = [p1, p2] self.deck = Deck() self.round_no = 0 def start(self): for p in self.players: p.draw_from(self.deck, 3) def one_turn(self, attacker, defender): attacker.draw_from(self.deck, 1) card = attacker.play_best() if card is None: print("%s 没牌可出，只能干瞪眼！" % attacker.name) return defender.take_damage(card.attack) print("%s 打出 %s，对 %s 造成 %d 点伤害（%s 剩余 HP %d）" % (attacker.name, card, defender.name, card.attack, defender.name, defender.hp)) def battle(self): print("=== 卡牌对战：%s VS %s ===" % (self.players[0].name, self.players[1].name)) self.start() p1, p2 = self.players winner = None while True: self.round_no += 1 print("—— 第 %d 回合（牌库剩 %d 张）——" % (self.round_no, len(self.deck))) for attacker, defender in ((p1, p2), (p2, p1)): self.one_turn(attacker, defender) if not defender.is_alive(): winner = attacker break if winner: break if len(self.deck) == 0 and all(len(p.hand) == 0 for p in self.players): print("双方都没牌了，握手言和，平局！") return None print("🏆 %s 获胜！大战 %d 回合" % (winner.name, self.round_no)) return winner if __name__ == "__main__": random.seed(42) game = Game(Player("勇者阿呆"), Player("BOSS·暗影领主")) game.battle()

    5⚡ 进阶挑战 · 三种隐藏难度

    挑战一 · 技能牌（继承）难度 ★★

    用继承扩展 Card：治疗牌不攻击，而是给自己回血。

    class HealCard(Card): # 继承 Card 的名字/攻击力/__str__ def __init__(self, name, heal): super().__init__(name, 0) # 攻击力 0 self.heal = heal # 新增：治疗量 def cast(self, player): # 新增：施放技能 player.hp += self.heal print("%s 使用 %s，回复 %d 点 HP！" % (player.name, self.name, self.heal))

    挑战二 · Buff 系统（状态）难度 ★★★

    给 Player 加 attack_buff 属性：打出牌时伤害 = 攻击力 + buff，每个回合结束后 buff 减 1。提示：在 Game 的回合循环末尾调用 player.tick_buff()。

    挑战三 · 简单 AI 对手（策略）难度 ★★★★

    给 BOSS 换个脑子：写一个 class SmartPlayer(Player)，重写 play_best()——当自己 HP 危急时才出大牌，优势时出小牌"省着打"（提示：if self.hp < 10: 出最大 else: 出最小）。让普通 Player 和 SmartPlayer 打 100 局统计胜率！
